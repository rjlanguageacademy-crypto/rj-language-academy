// ----- IELTS & TOEIC: exactly 100 interactive modules EACH -----
function rjxMakeExamModule(kind,n){
  const ieltsAreas=['Speaking Part 1','Speaking Part 2','Speaking Part 3','Writing Task 1','Writing Task 2','Reading','Listening','Band 7–9 Language','Academic Vocabulary','Grammar Range'];
  const toeicAreas=['Picture Description','Question–Response','Conversations','Talks & Announcements','Incomplete Sentences','Text Completion','Reading Comprehension','Business Vocabulary','Workplace Grammar','Speed & Accuracy'];
  const area=(kind==='IELTS'?ieltsAreas:toeicAreas)[(n-1)%10];
  const title=`${area} · Mastery ${String(n).padStart(3,'0')}`;
  const desc=`${kind} lesson ${n}: focused ${area.toLowerCase()} training with strategy, model language and a quick check.`;
  RJX_EXAM_BANK[title]={
    teach:`This ${kind} module develops ${area}. Work under exam-style pressure, identify exactly what the task requires, and choose language that is clear, accurate and appropriate. Module ${n} adds another layer of controlled practice so progress comes from repeated, varied exposure rather than one isolated exercise.`,
    strategy:['Read or listen for the exact task','Identify key words and likely distractors','Choose a clear structure before answering','Check accuracy and relevance','Review the explanation before continuing'],
    model:`Module ${n} model: answer the task directly, support it with a relevant detail, and keep the language controlled and natural.`,
    questions:[
      [`${kind} ${n}: what should you do first?`,['Identify the exact task and key information.','Guess before reading or listening.','Use memorised language without checking relevance.'],0,'Start by understanding the exact task.'],
      [`${kind} ${n} · ${area}: which response is strongest for this task?`,['A clear, relevant and accurate response.','A long response that does not answer the question.','A difficult response with avoidable grammar errors.'],0,'Relevance and control are more valuable than unnecessary complexity.'],
      [`${kind} ${n} · ${area}: what should you do with feedback before moving on?`,['Review mistakes and make one improved example.','Ignore feedback.','Repeat the same error deliberately.'],0,'Immediate review helps consolidate the skill.']
    ]
  };
  return [title,desc];
}

<script id="rjx-interactive-sounds">
/* RJ Language Academy sound system.
   All sounds are generated in-browser with Web Audio: no external sound files. */
(() => {
  let audioCtx = null;
  let lastOutcomeAt = 0;

  function ctx() {
    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
    if (!AudioContextClass) return null;
    if (!audioCtx) audioCtx = new AudioContextClass();
    if (audioCtx.state === 'suspended') audioCtx.resume().catch(() => {});
    return audioCtx;
  }

  function tone({freq=440, endFreq=null, type='sine', duration=0.08, gain=0.05, delay=0}) {
    const ac = ctx();
    if (!ac) return;
    const now = ac.currentTime + delay;
    const osc = ac.createOscillator();
    const amp = ac.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, now);
    if (endFreq) {
      osc.frequency.exponentialRampToValueAtTime(Math.max(1, endFreq), now + duration);
    }
    amp.gain.setValueAtTime(0.0001, now);
    amp.gain.exponentialRampToValueAtTime(gain, now + Math.min(0.015, duration / 4));
    amp.gain.exponentialRampToValueAtTime(0.0001, now + duration);
    osc.connect(amp);
    amp.connect(ac.destination);
    osc.start(now);
    osc.stop(now + duration + 0.03);
  }

  window.rjxPlayClick = function () {
    tone({freq: 560, endFreq: 390, type: 'sine', duration: 0.045, gain: 0.032});
  };

  /* Bell-like correct answer: bright fundamental plus two quiet overtones. */
  window.rjxPlayCorrect = function () {
    const now = performance.now();
    if (now - lastOutcomeAt < 90) return;
    lastOutcomeAt = now;
    tone({freq: 880, type: 'sine', duration: 0.42, gain: 0.085});
    tone({freq: 1760, type: 'sine', duration: 0.30, gain: 0.038, delay: 0.006});
    tone({freq: 2640, type: 'sine', duration: 0.19, gain: 0.020, delay: 0.012});
  };

  /* Wrong-answer buzzer: low, rough and clearly different from the bell. */
  window.rjxPlayWrong = function () {
    const now = performance.now();
    if (now - lastOutcomeAt < 90) return;
    lastOutcomeAt = now;
    tone({freq: 175, endFreq: 105, type: 'sawtooth', duration: 0.34, gain: 0.065});
    tone({freq: 118, endFreq: 82, type: 'square', duration: 0.29, gain: 0.030, delay: 0.035});
  };

  /* General click feedback only. Question handlers play their own bell/buzzer. */
  document.addEventListener('click', (event) => {
    const pressed = event.target.closest(
      'button, a, [role="button"], .click, .rjx-row, .rj-row, .rjx-member-level, .rj-member-tool, .rj-silver-card, .rj-silver-resource'
    );
    if (!pressed || pressed.disabled || pressed.getAttribute('aria-disabled') === 'true') return;
    window.rjxPlayClick();
  });
})();
</script>
