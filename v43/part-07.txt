function rjxPageControls(page,total,onPage){

  const wrap=document.createElement('div');wrap.className='rjx-bottom-nav';

  wrap.querySelector('[data-pg="prev"]')?.addEventListener('click',()=>onPage(page-1));
  wrap.querySelector('[data-pg="next"]')?.addEventListener('click',()=>onPage(page+1));
  return wrap;
}
