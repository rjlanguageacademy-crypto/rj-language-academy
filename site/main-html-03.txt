// V13: self-contained quiz feedback sounds. Web Audio keeps the single-file

let rjAudioContext;
function rjAudio(){
  const AudioContext=window.AudioContext||window.webkitAudioContext;
  if(!AudioContext) return null;
  if(!rjAudioContext) rjAudioContext=new AudioContext();
  if(rjAudioContext.state==='suspended') rjAudioContext.resume();
  return rjAudioContext;
}
function playCorrectSound(){
  const ctx=rjAudio(); if(!ctx) return;
  const now=ctx.currentTime;
  [[659.25,0,.34],[880,.13,.46]].forEach(([frequency,delay,end])=>{
    const oscillator=ctx.createOscillator(),gain=ctx.createGain();
    oscillator.type='sine'; oscillator.frequency.setValueAtTime(frequency,now+delay);
    gain.gain.setValueAtTime(.0001,now+delay);
    gain.gain.exponentialRampToValueAtTime(.22,now+delay+.018);
    gain.gain.exponentialRampToValueAtTime(.0001,now+end);
    oscillator.connect(gain).connect(ctx.destination);
    oscillator.start(now+delay); oscillator.stop(now+end+.02);
  });
}
function playWrongSound(){
  const ctx=rjAudio(); if(!ctx) return;
  const now=ctx.currentTime,oscillator=ctx.createOscillator(),gain=ctx.createGain();
  oscillator.type='sawtooth';
  oscillator.frequency.setValueAtTime(150,now);
  oscillator.frequency.linearRampToValueAtTime(92,now+.48);
  gain.gain.setValueAtTime(.0001,now);
  gain.gain.exponentialRampToValueAtTime(.14,now+.018);
  gain.gain.setValueAtTime(.14,now+.35);
  gain.gain.exponentialRampToValueAtTime(.0001,now+.52);
  oscillator.connect(gain).connect(ctx.destination);
  oscillator.start(now); oscillator.stop(now+.54);
}
const rjQuizSoundObserver=new MutationObserver(mutations=>{
  let correct=false,wrong=false;
  mutations.forEach(({target})=>{
    if(!(target instanceof Element)||!target.classList.contains('answer')) return;
    if(target.classList.contains('wrong')) wrong=true;
    else if(target.classList.contains('correct')) correct=true;
  });
  if(wrong) playWrongSound();

});
